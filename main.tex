\section*{מטרת הניסוי}

ניסוי זה בוחן את התנהגות הדיודה 1N4005 דרך מדידת האופיין שלה (עקומת ה-I-V, בהתאמה לנוסחת שוקלי),
ואת מעגל ה-RLD כמערכת דינמית לא-לינארית המפגינה מעבר לכאוס.

% Procedural description
% ----------------------------------------------------------------
\textbf{שבוע 1:}
בכל מדידה שינינו ידנית את האמפליטודה של מחולל האותות.
על מנת לחלץ את האופיין של הדיודה, השתמשנו בשתי שיטות:
בגל ריבועי, כאשר ממוצע ערכי המתח בחלקים היציבים של הגל שימש אותנו לחישוב הזרם לפי
$I_D=(V_{\text{Ch2}}-V_{\text{Ch1}})/R$;
ובגל משולש, אשר כיסה את טווח המתחים המלא. לאחר חישוב ממוצע נע חילצנו את עקומת ה-    $I(V)$.  
בכדי לאמוד את קיבול הדיודה, הוספנו offset שלילי למחולל האותות כדי לעבוד תחת ממתח אחורי,
ומעקומת הפריקה חילצנו את קבוע הזמן $\tau$, ומכאן $C_j=\tau/R$.

\textbf{שבוע 2:}
תחילה מצאנו את תדירות הרזוננס - כ-$\SI{34}{\kilo\hertz}$ - על ידי שינוי ידני של תדר
מחולל האותות עד שמתח הדיודה הגיע למקסימום.
לאחר מכן, בתדירות הרזוננס, שינינו ידנית את האמפליטודה בצעדים מ-200 mV עד 3570 mV;
בכל ערך המתנו שהמעגל יתייצב ולאחר מכן חילצנו את ערכי שיא המתח על הדיודה
לבניית מפת הביפורקציות הדיסקרטית.
לבניית המפה הרציפה, יצרנו אות AM עם גל נושא ב-$\SI{34}{\kilo\hertz}$,
תדר מודולציה של $\SI{0.7}{\hertz}$ ואמפליטודה מקסימלית של $\SI{7}{\volt}$ —
כך שהאמפליטודה עלתה וירדה במהלך מדידה אחת בת שנייה,
ואת שיאי המתח על הדיודה מדדנו לאורך כל הזמן.

% ----- Figure 1: Discrete I-V ----------------------------------------
\begin{figure}[H]
  \centering
  \includegraphics[width=\textwidth]{figures/graph1IV.png}
  \caption{%
    \textbf{עקומת I-V בדידה (גל ריבועי).}
    \caplabel{א}{תופעה}%
    עקומת מאפייני I-V של דיודת 1N4005 הנמדדת בגל ריבועי ($R = \SI{470}{\ohm}$).
    כל נקודה מייצגת ממוצע של דגימות בחלקים היציבים של הגל.
    נקודות עם $V_{pp} > \SI{2.0}{\volt}$ (6 נקודות: 2.4–10 V) אינן כלולות בהתאמה
    עקב התנגדות סדרתית בזרמים גבוהים.
    \caplabel{ב}{התאמה}%
    $I = I_S\!\left(e^{V_D/nV_T}-1\right)$,\quad $V_T = \SI{25.85}{\milli\volt}$.
    \caplabel{ג}{פרמטרים}%
    $n = 1.561 \pm 0.023$,\quad $I_S = (9.0 \pm 1.9)\times10^{-10}\,\si{\ampere}$.
    \caplabel{ד}{שגיאה}%
    קוונטיזציית ADC $\approx \SI{80}{\milli\volt}$ (2\,V/div, 8-bit);
    זרם: $\sigma_I = \sigma_{\Delta V}/\SI{470}{\ohm}$.
  }
  \label{fig:iv-discrete}
\end{figure}

% ----- Figure 2: Continuous I-V (triangle) ---------------------------
\begin{figure}[H]
  \centering
  \includegraphics[width=\textwidth]{figures/graph2IV_Triangle.png}
  \caption{%
    \textbf{עקומת I-V רציפה (גל משולש).}
    \caplabel{א}{תופעה}%
    עקומת מאפייני I-V של דיודת 1N4005 הנמדדת בגל משולש.
    הרצועה הכחולה הרחבה משקפת את שני כיווני הסריקה (עולה ויורדת).
    היסט חיובי קבוע של $\approx \SI{0.3}{\milli\ampere}$ בממתח אחורי
    הוא ארטיפקט של רמת בסיס ה-ADC, ולא הולכת דיודה פיזיקלית.
    \caplabel{ב}{התאמה}%
    $I = I_S\!\left(e^{V_D/nV_T}-1\right)$, מותאם לאזור ממתח קדמי ($V_D > \SI{0.3}{\volt}$).
    \caplabel{ג}{פרמטרים}%
    $n = 1.518$,\quad $I_S = 5.56\times10^{-10}\,\si{\ampere}$.
    שגיאות ההתאמה אינן מייצגות שכן מדובר בעקומה בודדת ממוצעת.
    \caplabel{ד}{שגיאה}%
    הרצועה הנראית לעין משקפת רעש; אין פס שגיאה פורמלי.
  }
  \label{fig:iv-triangle}
\end{figure}

% ----- Figure 3: C(V) ------------------------------------------------
\begin{figure}[H]
  \centering
  \includegraphics[width=\textwidth]{figures/graph3C_V.png}
  \caption{%
    \textbf{קיבול הדיודה $C_j(V)$.}
    \caplabel{א}{תופעה}%
    קיבול הדיודה 1N4005 פוחת עם עליית המתח תחת ממתח אחורי, בהתאם לתיאוריית שכבת הניגוד.
    סך הוריאציה הנמדדת $\approx \SI{4.5}{\pico\farad}$ על בסיס
    $\approx \SI{200}{\pico\farad}$, מה שמצביע על ריסון חלש של ההתאמה.
    \caplabel{ב}{התאמה}%
    $C_j(V) = a/\!\sqrt{|V|} + c$.
    \caplabel{ג}{פרמטרים}%
    $a = (6.06 \pm 0.64)\times10^{-12}\,\si{\farad\cdot\volt^{1/2}}$,\quad
    $c = 1.97\times10^{-10}\,\si{\farad}$.
    \caplabel{ד}{שגיאה}%
    אופקי: קוונטיזציית ADC;
    אנכי: $\sigma_C = \sigma_\tau/R$, כאשר $\sigma_\tau$ מוערך משאריות ההתפרקות ($\approx 10\%$).
  }
  \label{fig:cv}
\end{figure}
